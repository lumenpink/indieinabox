163    {

165
166        $lower = strtolower($basename);
167        if (!isset(self::$vaultFileIndex[$vaultRoot][$lower])) {
168            return null;
169        }
170
171        $matches = self::$vaultFileIndex[$vaultRoot][$lower];
172        if (count($matches) === 1) {
173            return $matches[0];
174        }
175
176        // Multiple files with the same name: rank by proximity to referencing file
177        $bestMatch = null;
178        $bestScore = -1;
179
180        $refParts = explode(DIRECTORY_SEPARATOR, trim($referencingFilePath, DIRECTORY_SEPARATOR));
181
182        foreach ($matches as $match) {

184            $score = 0;

186            for ($i = 0; $i < $len; $i++) {
187                if ($refParts[$i] === $matchParts[$i]) {
188                    $score++;

190                    break;
191                }
192            }
193
194            if ($score > $bestScore) {
195                $bestScore = $score;
196                $bestMatch = $match;
197            }
198        }
