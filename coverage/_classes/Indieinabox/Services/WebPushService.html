159    public function derToP1363(string $der): string
160    {
161        // DER sequence: 0x30 [len] 0x02 [rlen] [r] 0x02 [slen] [s]
162        $offset = 2;
163        if (ord($der[1]) > 0x80) {
164            $offset += (ord($der[1]) & 0x7f);
165        }
166
167        // Tag for r
168        $offset++; // skip 0x02
169        $rLen = ord($der[$offset++]);
170        $r = substr($der, $offset, $rLen);
171        $offset += $rLen;
172
173        // Tag for s
174        $offset++; // skip 0x02
175        $sLen = ord($der[$offset++]);
176        $s = substr($der, $offset, $sLen);
177
178        // Normalize r and s to exactly 32 bytes (strip leading null bytes if > 32, pad left with null bytes if < 32)
179        $r = ltrim($r, "\x00");
180        $s = ltrim($s, "\x00");
181
182        $r32 = str_pad($r, 32, "\x00", STR_PAD_LEFT);
