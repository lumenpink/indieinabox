60     */
61    public static function cleanBody(string $text): string
62    {
63        // Strip frontmatter if present
64        $text = (string) preg_replace('/^---[\s\S]*?---\s*/u', '', $text);
65        // Strip images and links but keep link text
66        $text = (string) preg_replace('/!\[.*?\]\(.*?\)/u', '', $text);
67        $text = (string) preg_replace('/\[(.*?)\]\(.*?\)/u', '$1', $text);
68        // Strip code blocks and inline code
69        $text = (string) preg_replace('/```[\s\S]*?```/u', '', $text);
70        $text = (string) preg_replace('/`.*?`/u', '', $text);
71        // Strip markdown formatting symbols
72        $text = (string) preg_replace('/(\*\*|__)(.*?)\1/s', '$2', $text);
73        $text = (string) preg_replace('/(\*|_)(.*?)\1/s', '$2', $text);
74        $text = (string) preg_replace('/~~(.*?)~~/s', '$1', $text);
75        // Strip HTML tags
76        $text = strip_tags($text);
77        // Strip headers, blockquotes, bullets
78        $text = (string) preg_replace('/^[#>\-*+]\s+/mu', '', $text);
79        // Normalize whitespace
80        $text = (string) preg_replace('/\s+/u', ' ', $text);
