578        async function interactPostMicropub(action, targetUrl) {
579            let content = '';
580            let payload = {
581                action: 'create',
582                h: 'entry'
583            };
584            
585            if (action === 'reply') {
586                content = prompt("Digite sua resposta para publicar no seu blog:");
587                if (!content) return;
588                
589                const item = window.timelineItems.find(i => i.url === targetUrl);
590                if (item) {
591                    let originalText = item.content.text || item.content.html || '';
592                    originalText = originalText.replace(/<[^>]+>/g, '').trim();
593                    if (originalText.length > 150) {
594                        originalText = originalText.substring(0, 150) + '...';
595                    }
596                    content += "\n\n> " + originalText + "\n>\n> -- [" + (item.author ? item.author.name : "Post Original") + "](" + item.url + ")";
597                }
598
599                payload['in-reply-to'] = targetUrl;
600                payload.content = content;
601                payload['mp-slug'] = 'reply-' + Date.now();
602            } else if (action === 'like') {
603                payload['like-of'] = targetUrl;
604                payload['mp-slug'] = 'like-' + Date.now();
605            } else if (action === 'repost') {
606                payload['repost-of'] = targetUrl;
607                payload['mp-slug'] = 'repost-' + Date.now();
608            }
609
610            try {
611                const formData = new URLSearchParams(payload);
612                const res = await fetch('/micropub', {
613                    method: 'POST',
614                    body: formData,
615                    credentials: 'same-origin',
616                    headers: {
617                        'Content-Type': 'application/x-www-form-urlencoded'
618                    }
619                });
620                
621                if (res.ok || res.status === 202) {
622                    alert(action.toUpperCase() + " publicado com sucesso no seu blog!");
623                } else {
624                    const text = await res.text();

626                }
627            } catch (err) {
628                console.error(err);
629                alert("Falha ao enviar interação.");
